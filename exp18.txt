import pandas as pd
import matplotlib.pyplot as plt
import scipy.stats as stats

# Create the dataset
data = {
    'age': [23, 23, 27, 27, 30, 41, 47, 49, 50,
            52, 54, 54, 56, 57, 58, 58, 60, 61],

    '%fat': [9.5, 26.5, 7.8, 17.8, 31.4, 25.9,
             27.4, 27.2, 31.2, 34.6, 42.5, 28.8,
             33.4, 30.2, 34.1, 32.9, 41.2, 35.7]
}

# Create DataFrame
df = pd.DataFrame(data)

print("AGE AND BODY FAT DATASET")
print(df)

# Calculate mean, median and standard deviation
print("\nSTATISTICAL MEASURES")

print("\nMean:")
print(df.mean())

print("\nMedian:")
print(df.median())

print("\nStandard Deviation:")
print(df.std())

# Draw Boxplots
plt.figure(figsize=(8, 5))

plt.boxplot([df['age'], df['%fat']],
            labels=['Age', 'Body Fat'])

plt.title("Boxplot of Age and Body Fat")
plt.ylabel("Values")
plt.show()

# Draw Scatter Plot
plt.figure(figsize=(8, 5))

plt.scatter(df['age'], df['%fat'])

plt.xlabel("Age")
plt.ylabel("Body Fat (%)")
plt.title("Scatter Plot of Age vs Body Fat")
plt.grid()
plt.show()

# Draw Q-Q Plot for Age
plt.figure(figsize=(6, 5))

stats.probplot(df['age'], dist="norm", plot=plt)

plt.title("Q-Q Plot of Age")
plt.show()

# Draw Q-Q Plot for Body Fat
plt.figure(figsize=(6, 5))

stats.probplot(df['%fat'], dist="norm", plot=plt)

plt.title("Q-Q Plot of Body Fat")
plt.show()
